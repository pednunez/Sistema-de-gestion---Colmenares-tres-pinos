@router.get("/catalogo", response_model=CatalogoMovilResponse)
def obtener_catalogo(
    db: Session = Depends(get_db),
    usuario: Usuario = Depends(obtener_usuario_actual),
):
    apiarios = db.scalars(
        select(Apiario)
        .where(Apiario.activo == True)
        .order_by(Apiario.nombre, Apiario.id)
    ).all()

    colmenas = db.scalars(
        select(Colmena)
        .where(Colmena.activo == True)
        .order_by(Colmena.codigo, Colmena.id)
    ).all()

    grupos = db.scalars(
        select(GrupoColmena)
        .where(GrupoColmena.activo == True)
        .order_by(GrupoColmena.apiario_id, GrupoColmena.nombre)
    ).all()

    miembros: dict[int, list[int]] = {grupo.id: [] for grupo in grupos}

    if grupos:
        filas = db.execute(
            select(GrupoColmenaAsignacion.grupo_id, Colmena.id)
            .join(Colmena, Colmena.id == GrupoColmenaAsignacion.colmena_id)
            .where(
                GrupoColmenaAsignacion.grupo_id.in_(list(miembros)),
                GrupoColmenaAsignacion.fecha_retiro.is_(None),
                Colmena.activo == True,
            )
            .order_by(Colmena.codigo, Colmena.id)
        ).all()

        for grupo_id, colmena_id in filas:
            miembros[grupo_id].append(colmena_id)

    tratamientos = db.scalars(
        select(Tratamiento)
        .where(Tratamiento.estado == "EN_CURSO")
        .order_by(Tratamiento.colmena_id, Tratamiento.fecha_inicio)
    ).all()

    return CatalogoMovilResponse(
        generado_en=datetime.now(timezone.utc),
        usuario=UsuarioCatalogo(id=usuario.id, nombre=usuario.nombre, rol=usuario.rol),
        apiarios=[
            ApiarioCatalogo(id=a.id, nombre=a.nombre, ubicacion=a.ubicacion)
            for a in apiarios
        ],
        colmenas=[
            ColmenaCatalogo(
                id=c.id,
                codigo=c.codigo,
                codigo_qr=c.codigo_qr,
                apiario_id=c.apiario_id,
                estado=c.estado,
                cantidad_marcos=c.cantidad_marcos,
            )
            for c in colmenas
        ],
        grupos=[
            GrupoCatalogo(
                id=g.id,
                apiario_id=g.apiario_id,
                nombre=g.nombre,
                colmena_ids=miembros[g.id],
            )
            for g in grupos
        ],
        tratamientos_en_curso=[
            TratamientoCatalogo(
                id=t.id,
                colmena_id=t.colmena_id,
                tipo_tratamiento=t.tipo_tratamiento,
                producto=t.producto,
                dosis=t.dosis,
                fecha_inicio=t.fecha_inicio,
            )
            for t in tratamientos
        ],

def valor_json(valor):
    """
    Convierte tipos especiales a valores compatibles con JSON.
    """

    if isinstance(valor, (datetime, date)):
        return valor.isoformat()

    if isinstance(valor, UUID):
        return str(valor)

    if isinstance(valor, Enum):
        return valor.value

    if isinstance(valor, Decimal):
        return float(valor)

    return valor


def tratamiento_a_dict(tratamiento):
    """
    Convierte automÃ¡ticamente todas las columnas
    del tratamiento a un diccionario.
    """

    return {
        columna.key: valor_json(
            getattr(tratamiento, columna.key)
        )
        for columna in sa_inspect(tratamiento).mapper.column_attrs
    }

# ==========================================================
#
# Si el id_cliente ya llegÃ³ antes (la app reintentÃ³ porque
# se cortÃ³ la seÃ±al), se devuelve el registro existente con
# 200 en vez de crear otro. Un id_cliente de otro usuario
# se rechaza: no se entrega un registro ajeno.
# ==========================================================

def registro_ya_sincronizado(registro, usuario: Usuario, response: Response):

    if registro.usuario_id != usuario.id:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="El id_cliente ya fue usado por otro usuario"
        )

    response.status_code = status.HTTP_200_OK

    return registro

@router.post(
    "/",
    response_model=TratamientoResponse,
    status_code=status.HTTP_201_CREATED
)
def crear_tratamiento(
    datos: TratamientoCreate,
    response: Response,
    db: Session = Depends(get_db),
    usuario_actual: Usuario = Depends(obtener_usuario_actual)
):

    # ------------------------------------------------------
    # REINTENTO DE SINCRONIZACIÃ“N
    # ------------------------------------------------------

    if datos.id_cliente is not None:

        existente = tratamiento_service.obtener_por_id_cliente(
            db,
            datos.id_cliente
        )

        if existente is not None:
            return registro_ya_sincronizado(existente, usuario_actual, response)

    # ------------------------------------------------------
    # VALIDAR COLMENA
    # ------------------------------------------------------

    colmena = tratamiento_service.obtener_colmena_activa(
        db,
        datos.colmena_id
    )

    if colmena is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="La colmena indicada no existe o estÃ¡ inactiva"
        )

    # ------------------------------------------------------
    # VALIDAR INSPECCIÃ“N SI FUE INDICADA
    # ------------------------------------------------------

    if datos.inspeccion_id is not None:

        inspeccion = tratamiento_service.obtener_inspeccion(
            db,
            datos.inspeccion_id
        )

        if inspeccion is None:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="La inspecciÃ³n indicada no existe o estÃ¡ inactiva"
            )

        if inspeccion.colmena_id != datos.colmena_id:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="La inspecciÃ³n no pertenece a la colmena indicada"
            )

    # ------------------------------------------------------
    # EL USUARIO REAL VIENE DEL JWT
    # ------------------------------------------------------

    datos_seguros = datos.model_copy(
        update={
            "usuario_id": usuario_actual.id
        }
    )

    try:
        tratamiento = tratamiento_service.crear_tratamiento(
            db,
            datos_seguros
        )
    except IntegrityError:
        db.rollback()

        # Dos envÃ­os simultÃ¡neos del mismo id_cliente: gana el primero.
        existente = (
            tratamiento_service.obtener_por_id_cliente(db, datos.id_cliente)
            if datos.id_cliente is not None
            else None
        )

        if existente is None:
            raise

        return registro_ya_sincronizado(existente, usuario_actual, response)

    # ------------------------------------------------------
    # AUDITORÃA
    # ------------------------------------------------------

    auditoria_service.registrar_auditoria(
        db,
        AuditoriaCreate(
            usuario_id=usuario_actual.id,
            entidad="TRATAMIENTO",
            entidad_id=tratamiento.id,
            accion="CREAR",
            datos_anteriores=None,
            datos_nuevos=tratamiento_a_dict(
                tratamiento
            )
        )
    )

    return tratamiento

@router.patch(
    "/{tratamiento_id}",
    response_model=TratamientoResponse
)
def actualizar_tratamiento(
    tratamiento_id: int,
    datos: TratamientoUpdate,
    db: Session = Depends(get_db),
    usuario_actual: Usuario = Depends(obtener_usuario_actual)
):

    tratamiento = tratamiento_service.obtener_tratamiento(
        db,
        tratamiento_id
    )

    if tratamiento is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Tratamiento no encontrado"
        )

    cambios = datos.model_dump(
        exclude_unset=True
    )

    # ------------------------------------------------------
    # FINALIZAR DESDE LA INSPECCIÃ“N
    # ------------------------------------------------------
    # Los apicultores rotan entre apiarios: cualquiera que
    # inspecciona la colmena puede finalizar un tratamiento
    # EN_CURSO, aunque lo haya registrado otra persona.
    # El resto de la ediciÃ³n queda para el autor o un ADMIN.
    # La auditorÃ­a registra quiÃ©n lo finalizÃ³.

    es_finalizacion = (
        set(cambios) == {"estado", "fecha_fin"}
        and cambios["estado"] == "FINALIZADO"
        and cambios["fecha_fin"] is not None
        and tratamiento.estado == "EN_CURSO"
    )

    if not (
        es_finalizacion
        and usuario_actual.rol == "APICULTOR"
    ):
        verificar_autor_o_admin(usuario_actual, tratamiento.usuario_id)

    if (

            raise HTTPException(

            )

    def denied(self,statement):
        savepoint=self.connection.begin_nested()
        try:
            with self.assertRaises(DBAPIError) as caught:
                self.connection.execute(text(statement))
            self.assertEqual(getattr(caught.exception.orig,'sqlstate',None),'42501')
        finally:
            savepoint.rollback()

        self.assertEqual(self.scalar('SELECT current_user'),'colmenares_app')
        flags=self.connection.execute(text('SELECT rolsuper,rolcreatedb,rolcreaterole,rolreplication,rolbypassrls FROM pg_roles WHERE rolname=current_user')).one()
        self.assertFalse(any(flags))
        self.assertFalse(self.scalar('SELECT EXISTS(SELECT 1 FROM pg_auth_members WHERE member=(SELECT oid FROM pg_roles WHERE rolname=current_user))'))
        self.assertFalse(self.scalar("SELECT EXISTS(SELECT 1 FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname='public' AND c.relowner=(SELECT oid FROM pg_roles WHERE rolname=current_user))"))
        self.assertFalse(self.scalar("SELECT has_schema_privilege(current_user,'public','CREATE')"))
        self.assertFalse(self.scalar('SELECT has_database_privilege(current_user,current_database(),\'CREATE\')'))

    def test_audit_insert_and_select_are_allowed_with_rollback(self):

        # The enclosing transaction is always rolled back, including on failure.

    def test_audit_update_is_denied(self):

    def test_audit_delete_is_denied(self):

          },
        }

  } catch (error) {

export class ErrorAPI extends Error {
  constructor(
    public readonly estado: number,
    mensaje: string
  ) {
    super(mensaje);
    this.name = 'ErrorAPI';
  }
}

type OpcionesPeticion = {
method?: 'GET' | 'POST' | 'PATCH' | 'DELETE';
  body?: unknown;
  token?: string;
};

// ============================================
// OBTENER MENSAJES DE ERROR DEL BACKEND
// ============================================

function obtenerDetalle(datos: unknown): string | null {
  if (
    !datos ||
    typeof datos !== 'object' ||
    !('detail' in datos)
  ) {
    return null;
  }

  const detalle = (datos as { detail: unknown }).detail;

  if (typeof detalle === 'string') {
    return detalle;
  }

  return null;
}

// ============================================
// REALIZAR PETICIONES A FASTAPI
// ============================================

export async function solicitarAPI<T>(
  ruta: string,
  opciones: OpcionesPeticion = {}
): Promise<T> {
  const controller = new AbortController();

  const temporizador = setTimeout(() => {
    controller.abort();
  }, API_TIMEOUT_MS);

  const headers: Record<string, string> = {
    Accept: 'application/json',
    'X-Correlation-ID': Crypto.randomUUID(),
  };

  // Enviar datos JSON cuando corresponda.
  if (opciones.body !== undefined) {
    headers['Content-Type'] = 'application/json';
  }

  // Incluir autenticaciÃ³n cuando exista un token.
  if (opciones.token) {
    headers.Authorization = `Bearer ${opciones.token}`;
  }

  try {
    const respuesta = await fetch(
      `${API_BASE_URL}${ruta}`,
      {
        method: opciones.method ?? 'GET',
        headers,
        body:
          opciones.body === undefined
            ? undefined
            : JSON.stringify(opciones.body),
        signal: controller.signal,
      }
    );

    const datos: unknown = await respuesta
      .json()
      .catch(() => null);

    // ========================================
    // MANEJO DE ERRORES HTTP
    // ========================================

    if (!respuesta.ok) {
      const detalle = obtenerDetalle(datos);

      let mensaje =
        detalle ??
        `Error del servidor (${respuesta.status}).`;

      if (respuesta.status === 401) {
        mensaje =
          'Credenciales incorrectas o sesiÃ³n vencida.';
      } else if (respuesta.status === 403) {
        mensaje =
          detalle ??
          'No tienes permisos para realizar esta acciÃ³n.';
      } else if (respuesta.status === 422) {
        mensaje =
          'El servidor rechazÃ³ los datos ingresados.';
      } else if (respuesta.status >= 500) {
        mensaje =
          'El servidor presentÃ³ un problema. Intenta nuevamente.';
      }

      throw new ErrorAPI(
        respuesta.status,
        mensaje
      );
    }

    if (datos === null) {
      throw new ErrorAPI(
        0,
        'El servidor no devolviÃ³ una respuesta JSON vÃ¡lida.'
      );
    }

    return datos as T;

  } catch (error) {
    if (error instanceof ErrorAPI) {
      throw error;
    }

    if (
      error instanceof Error &&
      error.name === 'AbortError'
    ) {
      throw new ErrorAPI(
        0,
        'La conexiÃ³n con FastAPI tardÃ³ demasiado.'
      );
    }

    throw new ErrorAPI(
      0,
      'No fue posible conectar con FastAPI. Comprueba la IP, el puerto y la conexiÃ³n.'
    );

  } finally {
    clearTimeout(temporizador);
  }
}

  const signos = convertirBooleano(
    detalle.signos_enfermedad
  );

  if (signos === null) {
    throw new Error(
      'No estÃ¡ definido el campo de signos de enfermedad.'
    );
  }

  return {
    colmena_id: colmena.id,
    usuario_id: usuarioId,
    id_cliente: inspeccion.id_cliente,
    fecha_inspeccion: inspeccion.fecha_local,

    estado_general: detalle.estado_general,

    reina_observada: convertirBooleano(
      detalle.reina_observada
    ),

    presencia_cria: convertirBooleano(
      detalle.presencia_cria
    ),

    postura: convertirBooleano(
      detalle.postura
    ),

    nivel_poblacion: detalle.nivel_poblacion,
    reservas_alimento: detalle.reservas_alimento,
    miel: detalle.nivel_miel,

    alimentacion_suministrada:
      detalle.alimentacion.trim() || null,

    signos_enfermedad: signos,

    enfermedad_observada:
      detalle.enfermedad_observada.trim() || null,

    tiras_medicamento:
      detalle.tiras_medicamento,

    tiene_alza: tieneAlza,
    cantidad_marcos: cantidadReal,
    marcos: marcosAPI,
    observaciones: inspeccion.observaciones,
  };
}

// ==========================================
// ENVIAR PENDIENTES A FASTAPI
// TODAVÃA NO SE LLAMA AUTOMÃTICAMENTE
// ==========================================

export async function sincronizarInspeccionesPendientes():
  Promise<ResultadoSincronizacion> {
  if (sincronizacionActiva) {
    throw new Error(
      'Ya hay una sincronizaciÃ³n de inspecciones en curso.'
    );
  }

  sincronizacionActiva = true;

  const resultado: ResultadoSincronizacion = {
    revisadas: 0,
    sincronizadas: 0,
    fallidas: 0,
    errores: [],
  };

  try {
    const sesion = await leerSesionGuardada();

    if (!sesion || !tokenVigente(sesion)) {
      throw new Error(
        'No existe una sesiÃ³n vÃ¡lida para sincronizar.'
      );
    }

    const catalogo = await leerCatalogoLocal(
      sesion.usuario.id
    );

    if (!catalogo) {
      throw new Error(
        'No hay un catÃ¡logo oficial para sincronizar.'
      );
    }

    // Inicializar migraciones de SQLite.
    await listarInspeccionesLocales();

    const db = await SQLite.openDatabaseAsync(
      'colmenas.db'
    );

    // Solo registros pendientes del usuario.
    // Los registros antiguos sin propietario
    // no se envÃ­an.
    const pendientes =
      await db.getAllAsync<InspeccionGuardada>(
        `SELECT *
         FROM inspecciones_locales
         WHERE estado = 'PENDIENTE'
           AND servidor_id IS NULL
           AND usuario_local_id = ?
         ORDER BY id ASC`,
        [sesion.usuario.id]
      );

    for (const inspeccion of pendientes) {
      // Volver a comprobar el vencimiento
      // durante sincronizaciones largas.
      if (!tokenVigente(sesion)) {
        resultado.errores.push({
          idLocal: inspeccion.id,
          mensaje: 'La sesiÃ³n venciÃ³ durante el envÃ­o.',
        });
        resultado.fallidas++;
        break;
      }

      resultado.revisadas++;

      try {
        const datos = await prepararInspeccion(
          db,
          inspeccion,
          sesion.usuario.id,
          catalogo
        );

        // El backend admite id_cliente para
        // reintentos seguros: 201 o 200.
        const respuesta =
          await solicitarAPI<RespuestaInspeccion>(
            '/inspecciones/',
            {
              method: 'POST',
              token: sesion.accessToken,
              body: datos,
            }
          );

        // No marcar Ã©xito sin comprobar
        // que la respuesta corresponde
        // al registro enviado.
        if (
          !Number.isSafeInteger(respuesta.id) ||
          respuesta.id <= 0 ||
          !respuesta.id_cliente ||
          normalizarCodigo(respuesta.id_cliente) !==
            normalizarCodigo(inspeccion.id_cliente) ||
          respuesta.colmena_id !== datos.colmena_id ||
          respuesta.usuario_id !== sesion.usuario.id
        ) {
          throw new Error(
            'La respuesta del servidor no coincide con la inspecciÃ³n enviada.'
          );
        }

        const cambio = await db.runAsync(
          `UPDATE inspecciones_locales
           SET estado = 'SINCRONIZADO',
               servidor_id = ?
           WHERE id = ?
             AND id_cliente = ?
             AND usuario_local_id = ?
             AND estado = 'PENDIENTE'
             AND servidor_id IS NULL`,
          [
            respuesta.id,
            inspeccion.id,
            inspeccion.id_cliente,
            sesion.usuario.id,
          ]
        );

        if (cambio.changes !== 1) {
          throw new Error(
            'El servidor confirmÃ³ el registro, pero SQLite no pudo marcarlo como sincronizado.'
          );
        }

        resultado.sincronizadas++;
      } catch (error) {
        resultado.fallidas++;

        resultado.errores.push({
          idLocal: inspeccion.id,
          mensaje:
            error instanceof Error
              ? error.message
              : 'Error desconocido de sincronizaciÃ³n.',
        });

        // Si falla la autorizaciÃ³n, no
        // insistir con los demÃ¡s registros.
        if (
          error instanceof ErrorAPI &&
          (
            error.estado === 401 ||
            error.estado === 403 ||
            error.estado === 409
          )
        ) {
          break;
        }

        // Otros errores dejan el registro
        // pendiente, sin borrar informaciÃ³n.
      }
    }

    return resultado;
  } finally {
    sincronizacionActiva = false;
  }
}
