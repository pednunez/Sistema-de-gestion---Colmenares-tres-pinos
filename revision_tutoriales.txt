// ==========================================
// FILTRAR MENU POR ROL
// ==========================================

function menuParaRol(rol: string): ItemMenu[] {
  return MENU.filter(item =>
    item.roles.includes(rol as Rol)
  );
}

// ==========================================
// IDENTIFICAR MODULO ACTIVO
// ==========================================

function obtenerModuloActivo(
  pantalla: Pantalla
): IdModulo {

  switch (pantalla) {

    case 'inicio':
      return 'dashboard';

    case 'apiarios-locales':
      return 'apiarios';

    case 'transferencias-locales':
      return 'transferencias';

    case 'stock-locales':
      return 'stock';

    case 'auditoria':
      return 'auditoria';

    case 'usuarios-locales':
      return 'usuarios';

    case 'colmenas-locales':
    case 'colmena':
      return 'colmenas';

    case 'revisiones-grupos':
      return 'grupos';

    case 'mis-inspecciones':
    case 'inspeccion':
      return 'inspecciones';

    case 'tratamientos-locales':
      return 'tratamientos';

    case 'escaner':
      return 'qr';

    default:
      return 'colmenas';
  }
}

// ==========================================
// VALIDAR QR OFICIAL
// ==========================================

const UUID =
  '[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}';

const PATRON_QR = new RegExp(
  `(?:^|/colmenas/qr/)(${UUID})/?(?:[?#].*)?$`,
  'i'
);

const PATRON_QR_WEB = new RegExp(
  `[?&]qr=(${UUID})(?:&|#|$)`,
  'i'
);

function obtenerUUIDDesdeQR(
  valor: string
): string | null {

  const texto = valor.trim();

  const coincidencia =
    texto.match(PATRON_QR) ??
    texto.match(PATRON_QR_WEB);

  if (!coincidencia) {
    return null;
  }

  return coincidencia[1].toLowerCase();
}

type EstadoAplicacion =
  | 'CARGANDO'
  | 'LOGIN'
  | 'AUTENTICADO'
  | 'SIN_CONEXION';

// ==========================================
// DESCARGAR CATALOGO DESDE FASTAPI
// ==========================================

async function actualizarCatalogo(
  sesion: SesionMovil
): Promise<string> {
  try {
    await descargarCatalogo(sesion);
    return '';

  } catch (error) {

    if (
      error instanceof ErrorAPI &&
      (error.estado === 401 || error.estado === 403)
    ) {
      throw error;
    }

    const catalogoLocal =
      await leerCatalogoLocal(
        sesion.usuario.id
      ).catch(() => null);

    if (
      error instanceof ErrorAPI &&
      error.estado === 404
    ) {
      return (
        'El servidor todavÃ­a no tiene disponible ' +
        '/movil/catalogo. Se necesita la versiÃ³n ' +
        'actualizada de la API.'
      );
    }

    if (catalogoLocal) {
      return (
        'No se pudo actualizar el catÃ¡logo. ' +
        'Se conserva la Ãºltima versiÃ³n guardada ' +
        'en SQLite.'
      );
    }

    return (
      'No se pudo descargar el catÃ¡logo oficial. ' +
      'TodavÃ­a no hay colmenas oficiales ' +
      'disponibles sin conexiÃ³n.'
    );
  }
}

function colmenaOperativa(
  colmena: ColmenaCatalogo
): boolean {
  return (
    colmena.estado !== 'INACTIVA' &&
    colmena.estado !== 'BAJA'
  );

function obtenerFecha(
  fecha?: string | null
): number {
  if (!fecha) return 0;

  const resultado = Date.parse(fecha);

  return Number.isFinite(resultado)
    ? resultado
    : 0;
}

function diasDesde(
  fecha?: string | null
): number | null {
  if (!fecha) return null;

  const partes = fecha
    .slice(0, 10)
    .split('-')
    .map(Number);

  let fechaInspeccion: Date;

  if (
    partes.length === 3 &&
    partes.every(Number.isFinite)
  ) {
    fechaInspeccion = new Date(
      partes[0],
      partes[1] - 1,
      partes[2]
    );
  } else {
    const fechaParseada = new Date(fecha);

    if (!Number.isFinite(fechaParseada.getTime())) {
      return null;
    }

    fechaInspeccion = new Date(
      fechaParseada.getFullYear(),
      fechaParseada.getMonth(),
      fechaParseada.getDate()
    );
  }

  const hoy = new Date();

  hoy.setHours(0, 0, 0, 0);

  fechaInspeccion.setHours(0, 0, 0, 0);

  return Math.floor(
    (hoy.getTime() - fechaInspeccion.getTime()) /
      86400000
  );
}

function ultimaInspeccionPorColmena(
  inspecciones: InspeccionDashboard[]
): Map<number, InspeccionDashboard> {

  const ultimas = new Map<
    number,
    InspeccionDashboard
  >();

  for (const inspeccion of inspecciones) {

    const actual = ultimas.get(
      inspeccion.colmena_id
    );

    if (
      !actual ||
      obtenerFecha(inspeccion.fecha_inspeccion) >
        obtenerFecha(actual.fecha_inspeccion)
    ) {
      ultimas.set(
        inspeccion.colmena_id,
        inspeccion
      );
    }
  }

  return ultimas;
}

function calcularAlertas(
  colmenas: ColmenaCatalogo[],
  inspecciones: InspeccionDashboard[]
): AlertaDashboard[] {

  const ultimas =
    ultimaInspeccionPorColmena(inspecciones);

  const alertas: AlertaDashboard[] = [];

  for (const colmena of colmenas) {

    if (!colmenaOperativa(colmena)) {
      continue;
    }

    const ultima = ultimas.get(colmena.id);

    const motivos: string[] = [];

    let prioridad: Prioridad | null = null;

    // REQUIERE REVISIÃ“N O CRÃTICO

    if (
      ultima?.estado_general ===
        'REQUIERE_REVISION' ||
      ultima?.estado_general === 'CRITICO'
    ) {

      motivos.push(
        ultima.estado_general === 'CRITICO'
          ? 'Ãšltima inspecciÃ³n en estado crÃ­tico'
          : 'Ãšltima inspecciÃ³n: requiere revisiÃ³n'
      );

      prioridad = 'alta';
    }

    // SIGNOS DE ENFERMEDAD

    const tieneSignos =
      ultima?.signos_enfermedad === true ||
      ultima?.signos_enfermedad === 1 ||
      ultima?.signos_enfermedad === '1' ||
      ultima?.signos_enfermedad === 'true';

    if (tieneSignos && ultima) {

      motivos.push(
        ultima.enfermedad_observada
          ? `Signos de enfermedad: ${ultima.enfermedad_observada}`
          : 'Signos de enfermedad'
      );

      prioridad = 'alta';
    }

    // ESTADOS DÃ‰BILES O REGULARES

    if (
      ultima?.estado_general === 'DEBIL' ||
      ultima?.estado_general === 'REGULAR'
    ) {

      motivos.push(
        ultima.estado_general === 'REGULAR'
          ? 'Ãšltima inspecciÃ³n en estado regular'
          : 'Ãšltima inspecciÃ³n: colmena dÃ©bil'
      );

      prioridad ??= 'media';
    }

    // EN OBSERVACIÃ“N

    if (colmena.estado === 'EN_OBSERVACION') {

      motivos.push('Marcada en observaciÃ³n');

      prioridad ??= 'media';
    }

    // SIN INSPECCIONES

    if (!ultima) {

      motivos.push('Sin inspecciones registradas');

      prioridad ??= 'media';

    } else {

      const dias = diasDesde(
        ultima.fecha_inspeccion
      );

      if (
        dias !== null &&
        dias > DIAS_SIN_INSPECCION
      ) {

        motivos.push(
          `Sin inspecciÃ³n hace ${dias} dÃ­as`
        );

        prioridad ??= 'media';
      }
    }

    if (motivos.length > 0 && prioridad) {

      alertas.push({
        colmena,
        prioridad,
        motivos,
        ultima,
      });
    }
  }

  return alertas.sort(
    (a, b) =>
      (a.prioridad === 'alta' ? 0 : 1) -
      (b.prioridad === 'alta' ? 0 : 1)
  );
}

import * as SecureStore from 'expo-secure-store';

import {
  ErrorAPI,
  solicitarAPI,
} from './cliente';

// ==========================================
// COLMENARES TRES PINOS
// AUTENTICACION MOVIL CON FASTAPI
// ==========================================

const CLAVE_SESION =
  'colmenares_tres_pinos_sesion_v1';

export type UsuarioMovil = {
  id: number;
  nombre: string;
  email: string;
  rol: string;
};

export type SesionMovil = {
  accessToken: string;
  expiraEn: number;
  usuario: UsuarioMovil;
};

type RespuestaLogin = {
  access_token: string;
  token_type: string;
  expira_en: number;
  usuario: UsuarioMovil;
};

// ==========================================
// VALIDACIONES
// ==========================================

const ROLES_PERMITIDOS = [
  'ADMIN',
  'APICULTOR',
];

function usuarioValido(
  valor: unknown
): valor is UsuarioMovil {
  if (
    !valor ||
    typeof valor !== 'object'
  ) {
    return false;
  }

  const usuario =
    valor as Partial<UsuarioMovil>;

  return (
    typeof usuario.id === 'number' &&
    Number.isInteger(usuario.id) &&
    usuario.id > 0 &&
    typeof usuario.nombre === 'string' &&
    usuario.nombre.trim().length > 0 &&
    typeof usuario.email === 'string' &&
    usuario.email.trim().length > 0 &&
    typeof usuario.rol === 'string' &&
    ROLES_PERMITIDOS.includes(usuario.rol)
  );
}

export function tokenVigente(
  sesion: SesionMovil
): boolean {
  // Margen de seguridad de 30 segundos.
  return Date.now() < sesion.expiraEn - 30000;
}

// ==========================================
// INICIAR SESION
// POST /auth/token
// ==========================================

export async function iniciarSesion(
  correoIngresado: string,
  password: string
): Promise<SesionMovil> {
  const email = correoIngresado
    .trim()
    .toLowerCase();

  // Validar correo.
  if (
    email.length < 5 ||
    email.length > 150 ||
    !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
  ) {
    throw new Error(
      'Ingresa un correo electrÃ³nico vÃ¡lido.'
    );
  }

  // Validar contraseÃ±a.
  if (
    password.length < 8 ||
    password.length > 128
  ) {
    throw new Error(
      'La contraseÃ±a debe tener entre 8 y 128 caracteres.'
    );
  }

  let respuesta: RespuestaLogin;

  try {
    respuesta =
      await solicitarAPI<RespuestaLogin>(
        '/auth/token',
        {
          method: 'POST',
          body: {
            email,
            password,
          },
        }
      );
  } catch (error) {
    if (
      error instanceof ErrorAPI &&
      error.estado === 404
    ) {
      throw new Error(
        'El servidor conectado no tiene disponible /auth/token. Debe utilizarse el backend actualizado del equipo.'
      );
    }

    throw error;
  }

  // ========================================
  // VALIDAR RESPUESTA DEL SERVIDOR
  // ========================================

  if (
    !respuesta ||
    typeof respuesta.access_token !==
      'string' ||
    respuesta.access_token.length === 0 ||
    respuesta.token_type?.toLowerCase() !==
      'bearer' ||
    !Number.isInteger(respuesta.expira_en) ||
    respuesta.expira_en <= 0 ||
    !usuarioValido(respuesta.usuario)
  ) {
    throw new Error(
      'El servidor devolviÃ³ una respuesta de autenticaciÃ³n invÃ¡lida.'
    );
  }

  // ========================================
  // CREAR SESION
  // ========================================

  const sesion: SesionMovil = {
    accessToken: respuesta.access_token,
    expiraEn:
      Date.now() +
      respuesta.expira_en * 1000,
    usuario: respuesta.usuario,
  };

  // Guardar token y datos de sesion
  // en almacenamiento seguro.
  await SecureStore.setItemAsync(
    CLAVE_SESION,
    JSON.stringify(sesion)
  );

  return sesion;
}

// ==========================================
// RECUPERAR SESION GUARDADA
// ==========================================

export async function leerSesionGuardada():
  Promise<SesionMovil | null> {
  const texto =
    await SecureStore.getItemAsync(
      CLAVE_SESION
    );

  if (!texto) {
    return null;
  }

  try {
    const datos: unknown =
      JSON.parse(texto);

    if (
      !datos ||
      typeof datos !== 'object'
    ) {
      return null;
    }

    const sesion =
      datos as Partial<SesionMovil>;

    if (
      typeof sesion.accessToken !==
        'string' ||
      !sesion.accessToken ||
      typeof sesion.expiraEn !==
        'number' ||
      !Number.isFinite(sesion.expiraEn) ||
      !usuarioValido(sesion.usuario)
    ) {
      return null;
    }

    return sesion as SesionMovil;
  } catch {
    return null;
  }
}

// ==========================================
// VERIFICAR USUARIO CON FASTAPI
// GET /auth/me
// ==========================================

export async function verificarUsuario(
  sesion: SesionMovil
): Promise<void> {
  if (!tokenVigente(sesion)) {
    throw new ErrorAPI(
      401,
      'El token ha vencido.'
    );
  }

  const usuario =
    await solicitarAPI<UsuarioMovil>(
      '/auth/me',
      {
        token: sesion.accessToken,
      }
    );

  if (
    !usuarioValido(usuario) ||
    usuario.id !== sesion.usuario.id
  ) {
    throw new ErrorAPI(
      401,
      'La identidad de la sesiÃ³n no coincide.'
    );
  }
}

import * as Crypto from 'expo-crypto';

import {
  API_BASE_URL,
  API_TIMEOUT_MS,
} from './config';

// ============================================
// COLMENARES TRES PINOS
// CLIENTE DE COMUNICACIÃ“N CON FASTAPI
// ============================================

export class ErrorAPI extends Error {
  constructor(
    public readonly estado: number,
    mensaje: string
  ) {
    super(mensaje);
    this.name = 'ErrorAPI';
  }
}

type OpcionesPeticion = {
method?: 'GET' | 'POST' | 'PATCH' | 'DELETE';
  body?: unknown;
  token?: string;
};

// ============================================
// OBTENER MENSAJES DE ERROR DEL BACKEND
// ============================================

function obtenerDetalle(datos: unknown): string | null {
  if (
    !datos ||
    typeof datos !== 'object' ||
    !('detail' in datos)
  ) {
    return null;
  }

  const detalle = (datos as { detail: unknown }).detail;

  if (typeof detalle === 'string') {
    return detalle;
  }

  return null;
}

// ============================================
// REALIZAR PETICIONES A FASTAPI
// ============================================

export async function solicitarAPI<T>(
  ruta: string,
  opciones: OpcionesPeticion = {}
): Promise<T> {
  const controller = new AbortController();

  const temporizador = setTimeout(() => {
    controller.abort();
  }, API_TIMEOUT_MS);

  const headers: Record<string, string> = {
    Accept: 'application/json',
    'X-Correlation-ID': Crypto.randomUUID(),
  };

  // Enviar datos JSON cuando corresponda.
  if (opciones.body !== undefined) {
    headers['Content-Type'] = 'application/json';
  }

  // Incluir autenticaciÃ³n cuando exista un token.
  if (opciones.token) {
    headers.Authorization = `Bearer ${opciones.token}`;
  }

  try {
    const respuesta = await fetch(
      `${API_BASE_URL}${ruta}`,
      {
        method: opciones.method ?? 'GET',
        headers,
        body:
          opciones.body === undefined
            ? undefined
            : JSON.stringify(opciones.body),
        signal: controller.signal,
      }
    );

    const datos: unknown = await respuesta
      .json()
      .catch(() => null);

    // ========================================
    // MANEJO DE ERRORES HTTP
    // ========================================

    if (!respuesta.ok) {
      const detalle = obtenerDetalle(datos);

      let mensaje =
        detalle ??
        `Error del servidor (${respuesta.status}).`;

      if (respuesta.status === 401) {
        mensaje =
          'Credenciales incorrectas o sesiÃ³n vencida.';
      } else if (respuesta.status === 403) {
        mensaje =
          detalle ??
          'No tienes permisos para realizar esta acciÃ³n.';
      } else if (respuesta.status === 422) {
        mensaje =
          'El servidor rechazÃ³ los datos ingresados.';
      } else if (respuesta.status >= 500) {
        mensaje =
          'El servidor presentÃ³ un problema. Intenta nuevamente.';
      }

      throw new ErrorAPI(
        respuesta.status,
        mensaje
      );
    }

    if (datos === null) {
      throw new ErrorAPI(
        0,
        'El servidor no devolviÃ³ una respuesta JSON vÃ¡lida.'
      );
    }

    return datos as T;

  } catch (error) {
    if (error instanceof ErrorAPI) {
      throw error;
    }

    if (
      error instanceof Error &&
      error.name === 'AbortError'
    ) {
      throw new ErrorAPI(
        0,
        'La conexiÃ³n con FastAPI tardÃ³ demasiado.'
      );
    }

    throw new ErrorAPI(
      0,
      'No fue posible conectar con FastAPI. Comprueba la IP, el puerto y la conexiÃ³n.'
    );

  } finally {
    clearTimeout(temporizador);
  }
}
