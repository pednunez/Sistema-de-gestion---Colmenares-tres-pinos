function fechaADate(valor: string): Date {
  const [a, m, d] = valor.split('-').map(Number);
  return new Date(a, m - 1, d, 12);
}

function dateAFecha(valor: Date): string {
  const a = valor.getFullYear();
  const m = String(
    valor.getMonth() + 1
  ).padStart(2, '0');

  const d = String(
    valor.getDate()
  ).padStart(2, '0');

  return `${a}-${m}-${d}`;
}

function moverAnios(
  valor: string,
  cantidad: number
): string {
  const fecha = fechaADate(valor);
  fecha.setFullYear(
    fecha.getFullYear() + cantidad
  );

  return dateAFecha(fecha);
}

type SelectorFechaProps = {
  titulo: string;
  valor: string | null;
  minimo: string;
  maximo: string;
  deshabilitado?: boolean;
  onChange: (valor: string | null) => void;
};

function SelectorFecha({
  titulo,
  valor,
  minimo,
  maximo,
  deshabilitado = false,
  onChange,
}: SelectorFechaProps) {
  const [abierto, setAbierto] = useState(false);
  const [temporal, setTemporal] =
    useState(new Date());

  const abrir = () => {
    if (deshabilitado) return;

    const inicial = valor || fechaLocalISO();

    const ajustada =
      inicial < minimo
        ? minimo
        : inicial > maximo
          ? maximo
          : inicial;

    setTemporal(fechaADate(ajustada));
    setAbierto(true);
  };

  const cambiar = (
    evento: { type: string },
    seleccion?: Date
  ) => {
    if (Platform.OS === 'android') {
      setAbierto(false);

      if (
        evento.type === 'set' &&
        seleccion
      ) {
        onChange(dateAFecha(seleccion));
      }
    } else if (seleccion) {
      setTemporal(seleccion);
    }
  };

  return (
    <View style={{ marginTop: 12 }}>
      <Text style={styles.label}>
        {titulo}
      </Text>

      <Pressable
        disabled={deshabilitado}
        style={[
          styles.input,
          deshabilitado &&
            styles.deshabilitado,
        ]}
        onPress={abrir}
      >
        <Text style={styles.texto}>
          ðŸ“… {valor ?? 'Seleccionar fecha'}
        </Text>
      </Pressable>

      {valor && !deshabilitado && (
        <Pressable
          onPress={() => onChange(null)}
        >
          <Text style={styles.enlace}>
            Quitar fecha
          </Text>
        </Pressable>
      )}

      {abierto &&
        Platform.OS === 'android' && (
          <DateTimePicker
            value={temporal}
            mode="date"
            minimumDate={
              fechaADate(minimo)
            }
            maximumDate={
              fechaADate(maximo)
            }
            onChange={cambiar}
          />
        )}

      {Platform.OS === 'ios' && (
        <Modal
          visible={abierto}
          transparent
          animationType="fade"
          onRequestClose={() =>
            setAbierto(false)
          }
        >
          <View style={styles.modalFondo}>
            <View style={styles.modalCaja}>
              <Text
                style={styles.tituloSeccion}
              >
                {titulo}
              </Text>

              <DateTimePicker
                value={temporal}
                mode="date"
                display="spinner"
                locale="es-CL"
                minimumDate={
                  fechaADate(minimo)
                }
                maximumDate={
                  fechaADate(maximo)
                }
                onChange={cambiar}
                style={{ height: 190 }}
              />

              <View style={styles.fila}>
                <Pressable
                  style={styles.botonBorde}
                  onPress={() =>
                    setAbierto(false)
                  }
                >
                  <Text
                    style={styles.textoBorde}
                  >
                    Cancelar
                  </Text>
                </Pressable>

                <Pressable
                  style={styles.boton}
                  onPress={() => {
                    onChange(
                      dateAFecha(temporal)
                    );
                    setAbierto(false);
                  }}
                >
                  <Text
                    style={styles.textoBoton}
                  >
                    Confirmar
                  </Text>
                </Pressable>
              </View>
            </View>
          </View>
        </Modal>
      )}
    </View>
  );
}

export type EstadoControlador =

  | 'SIN_CATALOGO'

export type ResultadoControlador = {

let ejecutando = false;

// ==========================================
// COMPROBAR INTERNET

function hayInternet(
  estado: NetInfoState
): boolean {
  return (
    estado.isConnected === true &&
    estado.isInternetReachable !== false
  );
}

// ==========================================
// EVALUAR Y SINCRONIZAR
// ==========================================

export async function ejecutarControlador():
  Promise<ResultadoControlador> {

  // Evitar ejecuciones simultÃ¡neas.

  if (ejecutando) {
    return {
      estado: 'EN_PROCESO',
      mensaje:
        'Ya se estÃ¡ comprobando la sincronizaciÃ³n.',
    };
  }

  ejecutando = true;

  try {

    // ======================================
    // 1. COMPROBAR CONECTIVIDAD
    // ======================================

    const red = await NetInfo.fetch();

    if (!hayInternet(red)) {
      return {
        estado: 'SIN_INTERNET',
        mensaje:
          'Sin conexiÃ³n. Los datos permanecen en SQLite.',
      };
    }

    // ======================================
    // 2. OBTENER SESIÃ“N
    // ======================================

    const sesion = await leerSesionGuardada();

    if (!sesion) {
      return {
        estado: 'SIN_SESION',
        mensaje:
          'No hay una sesiÃ³n iniciada. No se enviarÃ¡n registros.',
      };
    }

    // ======================================
    // 3. VERIFICAR TOKEN
    // ======================================

    if (!tokenVigente(sesion)) {
      return {
        estado: 'SESION_VENCIDA',
        mensaje:
          'El token venciÃ³. Las inspecciones pendientes se conservan.',
      };
    }

    // ======================================
    // 4. VERIFICAR CATÃLOGO
    // ======================================

    const catalogo = await leerCatalogoLocal(
      sesion.usuario.id
    );

    if (!catalogo) {
      return {
        estado: 'SIN_CATALOGO',
        mensaje:
          'No existe un catÃ¡logo oficial descargado para este usuario.',
      };
    }

    // ======================================
    // 5. COMPROBAR FASTAPI
    // ======================================

    try {

      const respuesta = await solicitarAPI<{
        status: string;
      }>(
        '/health',
        {
          method: 'GET',
        }
      );

      if (respuesta.status !== 'ok') {
        return {
          estado: 'SERVIDOR_NO_DISPONIBLE',
          mensaje:
            'El servidor no respondiÃ³ correctamente.',
        };
      }

    } catch {

      return {
        estado: 'SERVIDOR_NO_DISPONIBLE',
        mensaje:
          'No se pudo contactar con FastAPI.',
      };
    }

    // ======================================
    // 6. CONSULTAR REGISTROS SQLITE
    // ======================================

    const inspecciones =
      await listarInspeccionesLocales();

    // ======================================
    // 6.1 DIAGNÃ“STICO GENERAL
    // ======================================

    console.info(
      '[Colmenares] DIAGNOSTICO SQLITE:',
      {
        usuarioSesion: sesion.usuario.id,

        total: inspecciones.length,

        pendientes: inspecciones.filter(
          i => i.estado === 'PENDIENTE'
        ).length,

        pendientesDelUsuario: inspecciones.filter(
          i =>
            i.estado === 'PENDIENTE' &&
            i.usuario_local_id === sesion.usuario.id &&
            i.servidor_id === null
        ).length,

        sinPropietario: inspecciones.filter(
          i =>
            i.estado === 'PENDIENTE' &&
            i.usuario_local_id == null
        ).length,

        deOtroUsuario: inspecciones.filter(
          i =>
            i.estado === 'PENDIENTE' &&
            i.usuario_local_id != null &&
            i.usuario_local_id !== sesion.usuario.id
        ).length,

        conIdServidor: inspecciones.filter(
          i => i.servidor_id != null
        ).length,
      }
    );

    // ======================================
    // 6.2 DETALLE INDIVIDUAL DE REGISTROS
    // ======================================

    console.info(
      '[Colmenares] DETALLE DE INSPECCIONES LOCALES:',
      JSON.stringify(
        inspecciones.map(item => ({
          id: item.id,
          id_cliente: item.id_cliente,
          codigo_qr: item.codigo_qr,
          fecha: item.fecha_local,
          estado: item.estado,
          servidor_id: item.servidor_id,
          usuario_local_id: item.usuario_local_id,
        })),
        null,
        2
      )
    );

    // ======================================
    // 6.3 FILTRAR PENDIENTES DEL USUARIO
    // ======================================

    // Solo se consideran inspecciones:
    // - Pendientes.
    // - Sin identificador de servidor.
    // - Pertenecientes al usuario actual.

    const pendientes = inspecciones.filter(
      inspeccion =>
        inspeccion.estado === 'PENDIENTE' &&
        inspeccion.servidor_id === null &&
        inspeccion.usuario_local_id ===
          sesion.usuario.id
    );

    // ======================================
    // 6.4 COMPROBAR SI EXISTEN PENDIENTES
    // ======================================

    if (pendientes.length === 0) {
      return {
        estado: 'SIN_PENDIENTES',
        mensaje:
          'No hay inspecciones pendientes de este usuario.',
        pendientes: 0,
        sincronizadas: 0,
        fallidas: 0,
      };
    }

    // ======================================
    // 7. MODO DIAGNÃ“STICO
    // ======================================

    // No enviar registros mientras estÃ© desactivado.

    if (!HABILITAR_ENVIOS_REALES) {

    let estado: EstadoControlador;

    } else {

      estado = 'ERROR';
    }

    // ======================================
    // 10. INFORMAR ERRORES
    // ======================================

    if (resultado.errores.length > 0) {

      console.warn(

    // ======================================

    } else if (resultado.sincronizadas > 0) {

      mensaje =
        `SincronizaciÃ³n parcial: ${resultado.sincronizadas} enviadas y ${resultado.fallidas} fallidas.`;

    } else {

    }

    return {
      estado,

  } catch (error) {

    // ======================================
    // 12. CONTROL DE ERRORES
    // ======================================

    console.error(
      '[Colmenares] ERROR DEL CONTROLADOR:',

    return {

        error instanceof Error
          ? error.message
          : 'Error desconocido del controlador.',
    };

  } finally {

    // Liberar el bloqueo de ejecuciÃ³n.

    ejecutando = false;
  }
}

export function useControladorSincronizacion():
  void {

    // ======================================
    // EJECUTAR REVISIÃ“N
    // ======================================

    const revisar = async (
      motivo: string
    ) => {

      if (!activo) return;

      const resultado =
        await ejecutarControlador();

      if (!activo) return;

      console.info(
        '[Colmenares] Controlador:',
        motivo,
        resultado.estado,
        resultado.mensaje
      );
    };

    // ======================================
    // DETECTAR CUANDO REGRESA INTERNET
    // ======================================

    const cancelarRed =
      NetInfo.addEventListener(
        estado => {

          const conectado =
            hayInternet(estado);

          if (!conectado) {

            teniaInternet = false;

            return;
          }

          if (
            conectado &&
            !teniaInternet &&
            AppState.currentState === 'active'
          ) {

            teniaInternet = true;

            void revisar(
              'ConexiÃ³n disponible'
            );
          }
        }
      );

    // ======================================
    // VOLVER A COMPROBAR AL ABRIR LA APP
    // ======================================

    const suscripcionApp =
      AppState.addEventListener(
        'change',
        estado => {

          if (estado === 'active') {

            void revisar(
              'AplicaciÃ³n activa'
            );

          }
        }
      );

    // ======================================
    // LIMPIAR SUSCRIPCIONES
    // ======================================

    return () => {

      activo = false;

      cancelarRed();

      suscripcionApp.remove();
    };

  }, []);
}
